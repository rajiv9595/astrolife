// Fetch a Vedic birth chart using LifePath API
const response = await fetch('https://astrolife.onrender.com/compute', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-API-Key': 'sk_live_YOUR_API_KEY'
  },
  body: JSON.stringify({
    year: 1990, month: 5, day: 15,
    hour: 14, minute: 30, second: 0,
    tz: 'Asia/Kolkata',
    lat: 28.6139, lon: 77.2090
  })
});

const chart = await response.json();
console.log(chart.ascendant.sign);    // "Virgo"
console.log(chart.moon_sign);          // "Capricorn"
console.log(chart.planets.Sun.sign);   // Planet positions
console.log(chart.vimshottari);         // Dasha timeline
import requests

response = requests.post(
    "https://astrolife.onrender.com/compute",
    headers={
        "Content-Type": "application/json",
        "X-API-Key": "sk_live_YOUR_API_KEY"
    },
    json={
        "year": 1990, "month": 5, "day": 15,
        "hour": 14, "minute": 30, "second": 0,
        "tz": "Asia/Kolkata",
        "lat": 28.6139, "lon": 77.2090
    }
)

chart = response.json()
print(chart["ascendant"]["sign"])  # Virgo
print(chart["moon_sign"])           # Capricorn
print(chart["planets"])             # All planet data
curl -X POST https://astrolife.onrender.com/compute \
  -H "Content-Type: application/json" \
  -H "X-API-Key: sk_live_YOUR_API_KEY" \
  -d '{
    "year": 1990, "month": 5, "day": 15,

    "tz": "Asia/Kolkata",
    "lat": 28.6139, "lon": 77.2090
  }'

d10Dashamsha (D10) chart data
vimshottari.timelineFull Dasha periods
tithiTithi name & paksha
nakshatra_of_moonBirth nakshatra details
lucky_factorsLucky gems, colors, days
strengthsPlanet strength scores
